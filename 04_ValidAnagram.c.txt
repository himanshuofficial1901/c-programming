#include <stdbool.h>

bool isAnagram(char* s, char* t) {

    int count[256] = {0};

    int i = 0;

    while (s[i] != '\0') {
        count[(unsigned char)s[i]]++;
        i++;
    }

    i = 0;

    while (t[i] != '\0') {
        count[(unsigned char)t[i]]--;
        i++;
    }

    for (int j = 0; j < 256; j++) {
        if (count[j] != 0)
            return false;
    }

    return true;
}
